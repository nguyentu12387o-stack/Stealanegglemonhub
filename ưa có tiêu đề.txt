-- ═══════════ UI HELPERS ═══════════
local function mk(cls, parent, props)
    local i = New(cls)
    if props then for k, v in pairs(props) do pcall(function() i[k] = v end) end end
    if parent then pcall(function() i.Parent = parent end) end
    return i
end

local function corner(p, r)
    local c = New("UICorner")
    c.CornerRadius = UD(0, r or 6)
    c.Parent = p
end

local function stroke(p, col, thick, trans)
    local s = New("UIStroke")
    s.Color = col or C3(0, 220, 255)
    s.Thickness = thick or 1
    s.Transparency = trans or 0.4
    s.Parent = p
    return s
end

local function gradient(p, c1, c2, rot)
    local g = New("UIGradient")
    g.Color = ColorSequence.new(c1, c2)
    g.Rotation = rot or 0
    g.Parent = p
end

local function listLay(p, pad)
    local l = New("UIListLayout")
    l.Padding = UD(0, pad or 4)
    l.SortOrder = Enum.SortOrder.LayoutOrder
    l.Parent = p
end

local function gridLay(p, cols, w, h, px, py)
    local g = New("UIGridLayout")
    g.CellSize = UD2(0, w or 100, 0, h or 26)
    g.CellPadding = UD2(0, px or 4, 0, py or 4)
    g.SortOrder = Enum.SortOrder.LayoutOrder
    g.FillDirection = Enum.FillDirection.Horizontal
    g.FillDirectionMaxCells = cols or 2
    g.Parent = p
end

local function uiPad(p, t, l, r, b)
    local x = New("UIPadding")

    x.PaddingLeft = UD(0, l or 0)
    x.PaddingRight = UD(0, r or 0)
    x.PaddingBottom = UD(0, b or 0)
    x.Parent = p
end

local function CreateBVFly()
    if not S.hrp then return end
    DestroyFlyParts()
    S.flyBV = mk("BodyVelocity", S.hrp, {
        MaxForce = V3(1e6, 1e6, 1e6), P = 1e5, Velocity = V3z,
    })
    S.flyBG = mk("BodyGyro", S.hrp, {
        MaxTorque = V3(1e6, 1e6, 1e6), P = 1e4, CFrame = S.hrp.CFrame,
    })
end

local function CreateLVFly()
    if not S.hrp then return end
    DestroyFlyParts()
    S.flyAtc = mk("Attachment", S.hrp)
    S.flyLV = mk("LinearVelocity", S.hrp, {
        Attachment0 = S.flyAtc, MaxForce = 1e6,
        VelocityConstraintMode = Enum.VelocityConstraintMode.Vector,
        VectorVelocity = V3z,
    })
    S.flyAO = mk("AlignOrientation", S.hrp, {
        Attachment0 = S.flyAtc, MaxTorque = 1e6,
        Responsiveness = 200,
        Mode = Enum.OrientationAlignmentMode.OneAttachment,
    })
end

local function CreateFlyPartsForMode()
    DestroyFlyParts()
    if S.flyMode == "BodyVelocity" then
        CreateBVFly()
    elseif S.flyMode == "LinearVelocity" then
        CreateLVFly()
    elseif S.flyMode == "Camera" then
        S.flyBG = mk("BodyGyro", S.hrp, {
            MaxTorque = V3(1e6, 1e6, 1e6), P = 1e4, CFrame = S.hrp.CFrame,
        })
    end
end

local function StartFly()
    UpdateChar()
    if not S.hrp or not S.hum then 
        Notify("error", "Fly", "Không có nhân vật", 2); return 
    end
    S.fly = true
    EnableFlyNoclip()
    S.flyVel = V3z
    S.flyTarget = nil
    CreateFlyPartsForMode()
    
    if S.flyMode == "BodyVelocity" or S.flyMode == "LinearVelocity" or S.flyMode == "Camera" then
        pcall(function()
            S.hum.PlatformStand = true
            S.hum:SetStateEnabled(HS.Ragdoll, false)
            S.hum:SetStateEnabled(HS.FallingDown, false)
        end)
    end
    if VJGui then VJGui.Enabled = true end
    Notify("success", "✈️ FLY ON", "Mode: " .. S.flyMode, 2)
end

local function StopFly()
    S.fly = false
    DisableFlyNoclip()
    DestroyFlyParts()
    pcall(function() if S.hum then S.hum.PlatformStand = false end end)
    if VJGui then VJGui.Enabled = false end
    S.vjoy = 0
    S.flyTarget = nil
    Notify("warn", "✈️ FLY OFF", "", 2)
end

local function ToggleFly() if S.fly then StopFly() else StartFly() end end

    if not S.fly or not S.hrp or not S.hum then return end
    if S.hum.Health <= 0 then StopFly(); return end
    dt = m_min(dt, 0.05)

    local camCF = Camera.CFrame
    local look = camCF.LookVector
    local right = camCF.RightVector
    local fL = V3(look.X, 0, look.Z)
    local fR = V3(right.X, 0, right.Z)
    if fL.Magnitude > 0.01 then fL = fL.Unit end
    if fR.Magnitude > 0.01 then fR = fR.Unit end

    local spd = m_clamp(S.flySpd, 1, GetTier().maxSpd)
    local target = V3z

    if S.vjoyAct and m_abs(S.vjoy) > 0.05 then
        target = target + V3(0, 1, 0) * (S.vjoy * spd)
    end
    local move = S.hum.MoveDirection
    if move.Magnitude > 0.1 then
        target = target + (fL * move.Z + fR * move.X) * spd
    end

    local rate = (target.Magnitude > 0.1) and S.flyAccel or S.flyBrake
    local alpha = m_clamp(rate * dt, 0, 1)

    -- ══ MODE 1: CFrame ══
    if S.flyMode == "CFrame" then
        S.flyVel = S.flyVel:Lerp(target, alpha)
        if S.flyVel.Magnitude < 0.3 then S.flyVel = V3z end
        local newPos = S.hrp.Position + S.flyVel * dt
        if S.flyAltLock then newPos = V3(newPos.X, S.flyAltHeight, newPos.Z) end
        local bL = fL
        if S.flyVel.Magnitude > 5 then
            local hz = V3(S.flyVel.X, 0, S.flyVel.Z)
            if hz.Magnitude > 0.1 then bL = hz.Unit end
        end
        S.hrp.CFrame = CF(newPos, newPos + bL)
        S.hrp.AssemblyLinearVelocity = V3z
        S.hrp.AssemblyAngularVelocity = V3z

    -- ══ MODE 2: Camera ══
    elseif S.flyMode == "Camera" then
        S.flyVel = S.flyVel:Lerp(target, alpha)
        if S.flyVel.Magnitude < 0.3 then S.flyVel = V3z end
        local newPos = S.hrp.Position + S.flyVel * dt
        S.hrp.CFrame = CF(newPos, newPos + look)
        S.hrp.AssemblyLinearVelocity = V3z
        S.hrp.AssemblyAngularVelocity = V3z

    -- ══ MODE 3: BodyVelocity ══
    elseif S.flyMode == "BodyVelocity" then
        S.flyVel = S.flyVel:Lerp(target, alpha)
        if S.flyVel.Magnitude < 0.3 then S.flyVel = V3z end
        if not S.flyBV or not S.flyBV.Parent then CreateBVFly() end
        if S.flyBV then S.flyBV.Velocity = S.flyVel end
        if S.flyBG then S.flyBG.CFrame = camCF end

    -- ══ MODE 4: LinearVelocity ══
    elseif S.flyMode == "LinearVelocity" then
        S.flyVel = S.flyVel:Lerp(target, alpha)
        if S.flyVel.Magnitude < 0.3 then S.flyVel = V3z end
        if not S.flyLV or not S.flyLV.Parent then CreateLVFly() end
        if S.flyLV then S.flyLV.VectorVelocity = S.flyVel end
        if S.flyAO then S.flyAO.CFrame = camCF end

    -- ══ MODE 5: Behind ══
    elseif S.flyMode == "Behind" then
        local nearest, minDist = nil, m_huge
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local d = (hrp.Position - S.hrp.Position).Magnitude
                    if d < minDist then minDist = d; nearest = hrp end
                end
            end
        end
        if nearest then
            S.flyTarget = nearest
            local behindPos = nearest.Position - nearest.CFrame.LookVector * 8 + V3(0, 3, 0)
            local dir = behindPos - S.hrp.Position
            if dir.Magnitude > 0.5 then target = dir.Unit * spd end
        end
        local r2 = (target.Magnitude > 0.1) and S.flyAccel or S.flyBrake
        S.flyVel = S.flyVel:Lerp(target, m_clamp(r2 * dt, 0, 1))
        if S.flyVel.Magnitude < 0.3 then S.flyVel = V3z end
        local newPos = S.hrp.Position + S.flyVel * dt
        S.hrp.CFrame = CF(newPos, newPos + fL)
        S.hrp.AssemblyLinearVelocity = V3z
        S.hrp.AssemblyAngularVelocity = V3z
    end
end)

        Text = "↕", TextColor3 = C3RGB(255, 255, 255),
        TextSize = 20, Font = Enum.Font.GothamBold, ZIndex = 102,
    })

    local function upd(inp)
        if not VJBase.AbsoluteSize.Y or VJBase.AbsoluteSize.Y <= 0 then return end
        local c = VJBase.AbsolutePosition + VJBase.AbsoluteSize / 2
        local dY = inp.Position.Y - c.Y
        local m = (VJBase.AbsoluteSize.Y / 2) - 22
        local cl = m_clamp(dY, -m, m)
        S.vjoy = -cl / m
        VJThumb.Position = UD2(0.5, -22, 0.5, cl - 22)
    end
    local function rst()
        S.vjoy = 0; S.vjoyAct = false
        if VJThumb then VJThumb.Position = UD2(0.5, -22, 0.5, -22) end
    end

    VJBase.InputBegan:Connect(function(i)
        if i.UserInputType == UIT.Touch or i.UserInputType == UIT.MouseButton1 then
            VJTouch = i; S.vjoyAct = true; upd(i)
        end
    end)
    VJBase.InputEnded:Connect(function(i) if i == VJTouch then rst(); VJTouch = nil end end)
    UIS.InputChanged:Connect(function(i) if i == VJTouch then upd(i) end end)
    UIS.InputEnded:Connect(function(i) if i == VJTouch then rst(); VJTouch = nil end end)

-- ═══════════ LOOPS (TỐI ƯU — Chỉ chạy khi cần) ═══════════

-- Loop 1: Chung (0.5s/lần) — UpdateChar + Speed + Noclip + God + NoFall
spawn_t(function()
    while true do
        wait_t(0.5)
        pcall(function()
            UpdateChar()
            if S.hum and S.hum.Health > 0 then
                if S.hum.WalkSpeed ~= S.walkSpd then S.hum.WalkSpeed = S.walkSpd end
                if S.hum.JumpPower ~= S.jumpPow then S.hum.JumpPower = S.jumpPow end
                if S.godmode and S.hum.Health < S.hum.MaxHealth then S.hum.Health = S.hum.MaxHealth end
                if S.noclip and S.char then
                    for _, p in ipairs(S.char:GetDescendants()) do
                        if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end
                    end
                end
                if S.noFall then S.hum:SetStateEnabled(HS.FallingDown, false) end
            end
        end)
    end
end)

-- Loop 7: Combat (0.3s/lần) — KillAura + Hitbox
spawn_t(function()
    while true do
        wait_t(0.3)
        if S.killAura and S.hrp then
            pcall(function()
                for _, p in ipairs(Players:GetPlayers()) do
                    if p ~= LP and p.Character then
                        local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                        local hum = p.Character:FindFirstChildOfClass("Humanoid")
                        if hrp and hum and hum.Health > 0 then
                            if (hrp.Position - S.hrp.Position).Magnitude <= S.killAuraRange then
                                hum.Health = 0
                            end
                        end
                    end
                end
            end)
        end
        if S.hitbox then
            pcall(function()
                for _, p in ipairs(Players:GetPlayers()) do
                    if p ~= LP and p.Character then
                        local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                        if hrp then
                            hrp.Size = V3(S.hitboxSize, S.hitboxSize, S.hitboxSize)
                            hrp.Transparency = 1
                            hrp.CanCollide = false
                        end
                    end
                end
            end)
        end
    end
end)

-- Loop 8: Aimbot (chỉ khi bật)
spawn_t(function()
    while true do
        wait_t(0.1)
        if S.aimbot then
            pcall(function()
                local closest, minD = nil, S.aimRadius
                for _, p in ipairs(Players:GetPlayers()) do
                    if p ~= LP and p.Character then
                        local tp = p.Character:FindFirstChild(S.aimPart)
                        local hum = p.Character:FindFirstChildOfClass("Humanoid")
                        if tp and hum and hum.Health > 0 and S.hrp then
                            local d = (tp.Position - S.hrp.Position).Magnitude
                            if d < minD then minD = d; closest = tp end
                        end
                    end
                end
                if closest then
                    Camera.CFrame = CF(Camera.CFrame.Position, closest.Position)
                end
            end)
        end
    end
end)

-- Loop 9: ESP (chỉ khi bật, 1s/lần)
spawn_t(function()
    while true do
        wait_t(1)
        if S.esp or S.espNames then
            RefreshESP()
        end

    elseif i.KeyCode == KS.P then
        ToggleFly()
    elseif i.KeyCode == KS.K then
        if closeMenu then closeMenu() end
        if S.fly then StopFly() end
        if VJGui then VJGui.Enabled = false end
    elseif i.KeyCode == KS.Space then
        if S.fly then S.vjoyAct = true; S.vjoy = 1 end
    elseif i.KeyCode == KS.LeftShift then
        if S.fly then S.vjoyAct = true; S.vjoy = -1 end
    end
end)

    if gp then return end
    if i.KeyCode == KS.Space or i.KeyCode == KS.LeftShift then
        if S.fly then S.vjoyAct = false; S.vjoy = 0 end
    end
end)

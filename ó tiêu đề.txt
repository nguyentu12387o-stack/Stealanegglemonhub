local function ContainsKeyword(str, keywordList)
    if type(str) ~= "string" then return false end
    local lower = string_lower(str)
    for i = 1, #keywordList do
        if string_find(lower, keywordList[i], 1, true) then
            return true
        end
    end
    return false
end

-- ─── Tầng 1: Hook __namecall ───
local function InstallNamecallHook()
    if not getrawmetatable or not setreadonly or not hookmetamethod or not newcclosure then
        return false
    end
    local ok = pcall(function()
        local mt = getrawmetatable(game)
        setreadonly(mt, false)
        local original = mt.__namecall
        DefenseHooks.namecall = original

        mt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()

            if method == "Kick" then
                if typeof(self) == "Instance" and self:IsA("Player") then
                    DefenseState.KICK_BLOCKED = DefenseState.KICK_BLOCKED + 1
                    DLog("🚫 Chặn Kick() #" .. DefenseState.KICK_BLOCKED)
                    return nil
                end
            end

            if method == "Teleport" or method == "TeleportAsync" or method == "TeleportToPlaceInstance" then
                if typeof(self) == "Instance" and self:IsA("TeleportService") then
                    DefenseState.TELEPORT_BLOCKED = DefenseState.TELEPORT_BLOCKED + 1
                    DLog("🚫 Chặn Teleport #" .. DefenseState.TELEPORT_BLOCKED)
                    return nil
                end
            end

            if method == "FireServer" or method == "InvokeServer" then
                if typeof(self) == "Instance" then
                    if ContainsKeyword(self.Name, SUSPICIOUS_REMOTE_KEYWORDS) then
                        DLog("🚫 Chặn Remote: " .. self.Name)
                        return nil
                    end
                end
            end

            if method == "Destroy" then
                if typeof(self) == "Instance" then
                    if self == CoreGui then return nil end
                    if self:IsA("ScreenGui") and string_find(self.Name, "BlackZ") then
                        return nil
                    end
                end
            end

            return original(self, ...)
        end)
        setreadonly(mt, true)
    end)
    return ok
end

local function mk(cls, parent, props)
    local i = Instance_new(cls)
    if props then
        for k, v in pairs(props) do
            pcall(function() i[k] = v end)
        end
    end
    if parent then pcall(function() i.Parent = parent end) end
    return i
end

local function corner(p, r)
    local c = Instance_new("UICorner")
    pcall(function() c.CornerRadius = UDim_new(0, r or 6); c.Parent = p end)
    return c
end

local function stroke(p, col, thick, trans)
    local s = Instance_new("UIStroke")
    pcall(function()
        s.Color = col or Color3_new(1, 1, 1)
        s.Thickness = thick or 1
        s.Transparency = trans or 0
        s.Parent = p
    end)
    return s
end

local function gradient(p, c1, c2, rot)
    local g = Instance_new("UIGradient")
    pcall(function()
        g.Color = ColorSequence.new(c1, c2)
        g.Rotation = rot or 0
        g.Parent = p
    end)
    return g
end

local function listLayout(p, pad)
    local l = Instance_new("UIListLayout")
    pcall(function()
        l.Padding = UDim_new(0, pad or 4)
        l.SortOrder = Enum.SortOrder.LayoutOrder
        l.Parent = p
    end)
    return l
end

local function gridLayout(p, cols, cellW, cellH, padX, padY)
    local g = Instance_new("UIGridLayout")
    pcall(function()
        g.CellSize = UDim2_new(0, cellW or 100, 0, cellH or 26)
        g.CellPadding = UDim2_new(0, padX or 4, 0, padY or 4)
        g.SortOrder = Enum.SortOrder.LayoutOrder
        g.FillDirection = Enum.FillDirection.Horizontal
        g.FillDirectionMaxCells = cols or 2
        g.Parent = p
    end)
    return g
end

local function uiPadding(p, t, l, r, b)
    local x = Instance_new("UIPadding")
    pcall(function()
        x.PaddingTop = UDim_new(0, t or 0)
        x.PaddingLeft = UDim_new(0, l or 0)
        x.PaddingRight = UDim_new(0, r or 0)
        x.PaddingBottom = UDim_new(0, b or 0)
        x.Parent = p
    end)
    return x
end

local function tween(obj, time, props)
    pcall(function()
        TweenService:Create(obj, TweenInfo.new(time or 0.2), props):Play()
    end)
end

-- ─── 1. LOW GRAPHICS ───
local function SetLowGraphics(on)
    LF.lowGraphics = on
    pcall(function()
        if on then
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
            Lighting.Brightness = 1
            Lighting.GlobalShadows = false
            Lighting.FogEnd = 100000
            Lighting.EnvironmentDiffuseScale = 0
            Lighting.EnvironmentSpecularScale = 0
        else
            if LF.originalValues.quality then
                settings().Rendering.QualityLevel = LF.originalValues.quality
            end
            Lighting.Brightness = LF.originalValues.brightness
            Lighting.GlobalShadows = LF.originalValues.globalShadows
        end
    end)
end

-- ─── 2. REMOVE TEXTURES ───
local function RemoveTextures(on)
    LF.removeTextures = on
    pcall(function()
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("Decal") or obj:IsA("Texture") then
                if on then
                    if obj.Transparency ~= 1 then
                        obj:SetAttribute("BlackZ_OldTransparency", obj.Transparency)
                        obj.Transparency = 1
                    end
                else
                    local old = obj:GetAttribute("BlackZ_OldTransparency")
                    if old then
                        obj.Transparency = old
                        obj:SetAttribute("BlackZ_OldTransparency", nil)
                    end
                end
            end
        end
    end)
end

-- ─── 3. REMOVE FOG ───
local function RemoveFog(on)
    LF.removeFog = on
    pcall(function()
        if on then
            Lighting.FogEnd = 100000
            Lighting.FogStart = 100000
            Lighting.FogColor = Color3_RGB(0, 0, 0)
        else
            Lighting.FogEnd = LF.originalValues.fogEnd
            Lighting.FogStart = LF.originalValues.fogStart
        end
    end)
end

-- ─── 4. REMOVE PARTICLES ───
local function RemoveParticles(on)
    LF.removeParticles = on
    pcall(function()
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("ParticleEmitter") or obj:IsA("Trail")
                or obj:IsA("Smoke") or obj:IsA("Fire")
                or obj:IsA("Sparkles") or obj:IsA("Beam") then
                if on then
                    if obj.Enabled then
                        obj:SetAttribute("BlackZ_WasEnabled", true)
                        obj.Enabled = false
                    end
                else
                    if obj:GetAttribute("BlackZ_WasEnabled") then
                        obj.Enabled = true
                        obj:SetAttribute("BlackZ_WasEnabled", nil)
                    end
                end
            end
        end
    end)
end

ApplyInvisible = function(on)
    if not State.char then
        UpdateChar()
        if not State.char then return end
    end
    pcall(function()
        for _, v in ipairs(State.char:GetDescendants()) do
            if v:IsA("BasePart") then
                v.Transparency = on and 1 or 0
                v.LocalTransparencyModifier = on and 1 or 0
                v.CastShadow = not on
            elseif v:IsA("Decal") or v:IsA("Texture") then
                v.Transparency = on and 1 or 0
            elseif v:IsA("BillboardGui") or v:IsA("SurfaceGui") then
                v.Enabled = not on
            elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke")
                or v:IsA("Fire") or v:IsA("Sparkles") or v:IsA("PointLight")
                or v:IsA("SpotLight") then
                v.Enabled = not on
            elseif v:IsA("Accessory") then
                local handle = v:FindFirstChild("Handle")
                if handle and handle:IsA("BasePart") then
                    handle.Transparency = on and 1 or 0
                    handle.LocalTransparencyModifier = on and 1 or 0
                end
            end
        end
        if State.hum then
            if on then
                State.hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
                State.hum.NameDisplayDistance = 0
                State.hum.HealthDisplayDistance = 0
                State.hum.NameOcclusion = Enum.NameOcclusion.OccludeAll
                State.hum.HealthDisplayType = Enum.HumanoidHealthDisplayType.AlwaysOff
            else
                State.hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.Viewer
                State.hum.NameDisplayDistance = 100
                State.hum.HealthDisplayDistance = 100
                State.hum.NameOcclusion = Enum.NameOcclusion.NoOcclusion
                State.hum.HealthDisplayType = Enum.HumanoidHealthDisplayType.DisplayWhenDamaged
            end
        end
    end)
end

        State.espObjects[player.Name .. "_lbl"] = lbl
        State.espObjects[player.Name .. "_bb"] = bb
    end
end

RefreshESP = function()
    if not State.esp and not State.espNames then
        ClearESP()
        return
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            if not State.espObjects[player] and not State.espObjects[player.Name .. "_lbl"] then
                CreateESPForPlayer(player)
            end
        end
    end
    for key, obj in pairs(State.espObjects) do
        if typeof(key) == "Instance" and not key.Character then
            pcall(function() obj:Destroy() end)
            State.espObjects[key] = nil
        end
    end
end

task_spawn(function()
    while true do
        task_wait(1)
        RefreshESP()
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LP and player.Character then
                local lbl = State.espObjects[player.Name .. "_lbl"]
                local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if hrp and State.hrp and lbl then
                    local dist = math_floor((hrp.Position - State.hrp.Position).Magnitude)
                    local hp = hum and math_floor(hum.Health) or 0
                    lbl.Text = player.Name .. "\n[" .. hp .. " HP] " .. dist .. "m"
                end
            end
        end
    end
end)

local flyNoclipConn = nil

local function EnableFlyNoclip()
    if flyNoclipConn then return end
    flyNoclipConn = RunService.Stepped:Connect(function()
        if not State.flyEnabled or not State.char then return end
        for _, p in ipairs(State.char:GetDescendants()) do
            if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end
        end
    end)
end

local function DisableFlyNoclip()
    if flyNoclipConn then
        pcall(function() flyNoclipConn:Disconnect() end)
        flyNoclipConn = nil
    end
end

local function DestroyFlyParts()
    if State.flyBodyVel then pcall(function() State.flyBodyVel:Destroy() end); State.flyBodyVel = nil end
    if State.flyBodyGyro then pcall(function() State.flyBodyGyro:Destroy() end); State.flyBodyGyro = nil end
    if State.flyLinearVel then pcall(function() State.flyLinearVel:Destroy() end); State.flyLinearVel = nil end
    if State.flyAlignOrient then pcall(function() State.flyAlignOrient:Destroy() end); State.flyAlignOrient = nil end
end

local function CreateBodyVelocityFly()
    if not State.hrp then return end
    DestroyFlyParts()
    State.flyBodyVel = mk("BodyVelocity", State.hrp, {
        Name = "BlackZ_FlyVel",
        MaxForce = Vector3_new(1e6, 1e6, 1e6),

        MaxTorque = Vector3_new(1e6, 1e6, 1e6),

end

local function CreateLinearVelocityFly()
    if not State.hrp then return end
    DestroyFlyParts()

    State.flyLinearVel = mk("LinearVelocity", State.hrp, {
        Name = "BlackZ_LinearVel",
        Attachment0 = attach,
        MaxForce = 1e6,
        VelocityConstraintMode = Enum.VelocityConstraintMode.Vector,

        Responsiveness = 200,

StartFly = function()
    if not State.hrp or not State.hum then
        UpdateChar()
        if not State.hrp or not State.hum then
            Notify("error", "Fly", "Không tìm thấy nhân vật", 2)
            return
        end
    end
    State.flyEnabled = true
    EnableFlyNoclip()
    State.flyCurrentVel = Vector3_zero

    if State.flyMode ~= "CFrameBypass" and State.flyMode ~= "CFrame" and State.flyMode ~= "Tween" then
        if State.flyMode == "BodyVelocity" or State.flyMode == "Camera" or State.flyMode == "Behind" then
            CreateBodyVelocityFly()
        elseif State.flyMode == "LinearVelocity" then
            CreateLinearVelocityFly()
        end
        pcall(function()
            State.hum.PlatformStand = true
            State.hum:SetStateEnabled(HS.Ragdoll, false)
            State.hum:SetStateEnabled(HS.FallingDown, false)
        end)
    end

    if State.vjoyEnabled then
        if not VJoyGui then CreateVJoystick() end
        if VJoyGui then VJoyGui.Enabled = true end
    end
    Notify("success", "✈️ FLY ON", "Mode: " .. State.flyMode, 2)
end

StopFly = function()
    State.flyEnabled = false
    DisableFlyNoclip()
    DestroyFlyParts()
    pcall(function()
        if State.hum then State.hum.PlatformStand = false end
    end)
    if VJoyGui then VJoyGui.Enabled = false end
    State.vjoyValue = 0
    Notify("warn", "✈️ FLY OFF", "", 2)
end

ToggleFly = function()
    if State.flyEnabled then StopFly() else StartFly() end
end

RunService.RenderStepped:Connect(function(dt)
    if not State.flyEnabled then return end
    if not State.hrp or not State.hum then return end
    if State.hum.Health <= 0 then StopFly() return end

    dt = math_min(dt, 0.05)
    local camCF = Camera.CFrame
    local lookVec = camCF.LookVector
    local rightVec = camCF.RightVector

    local flatLook = Vector3_new(lookVec.X, 0, lookVec.Z)
    local flatRight = Vector3_new(rightVec.X, 0, rightVec.Z)
    if flatLook.Magnitude > 0.01 then flatLook = flatLook.Unit end
    if flatRight.Magnitude > 0.01 then flatRight = flatRight.Unit end

    local spd = math_clamp(State.flySpeed, 1, GetTierConfig().maxFlySpeed)
    local target = Vector3_zero

    if State.vjoyActive and math_abs(State.vjoyValue) > 0.05 then
        target = target + Vector3_new(0, 1, 0) * (State.vjoyValue * spd)
    end

    local move = State.hum.MoveDirection
    if move.Magnitude > 0.1 then
        target = target + (flatLook * move.Z + flatRight * move.X) * spd
    end

    if State.hjoyActive and State.hjoyValue.Magnitude > 0.1 then
        local dir = (flatRight * State.hjoyValue.X + flatLook * -State.hjoyValue.Y)
        if dir.Magnitude > 0.01 then
            target = target + dir.Unit * spd
        end
    end

    if State.flyMode == "BodyVelocity" or State.flyMode == "Camera" then
        local rate = (target.Magnitude > 0.1) and State.flyAccel or State.flyBrake
        local alpha = math_clamp(rate * dt, 0, 1)
        State.flyCurrentVel = State.flyCurrentVel:Lerp(target, alpha)
        if State.flyCurrentVel.Magnitude < 0.3 then State.flyCurrentVel = Vector3_zero end
        if not State.flyBodyVel or not State.flyBodyVel.Parent then CreateBodyVelocityFly() end
        if State.flyBodyVel then State.flyBodyVel.Velocity = State.flyCurrentVel end
        if State.flyBodyGyro then State.flyBodyGyro.CFrame = camCF end
        if State.flyMode == "Camera" then
            State.hrp.CFrame = CFrame_new(State.hrp.Position, State.hrp.Position + lookVec)
        end

    elseif State.flyMode == "LinearVelocity" then
        local rate = (target.Magnitude > 0.1) and State.flyAccel or State.flyBrake
        local alpha = math_clamp(rate * dt, 0, 1)
        State.flyCurrentVel = State.flyCurrentVel:Lerp(target, alpha)
        if State.flyCurrentVel.Magnitude < 0.3 then State.flyCurrentVel = Vector3_zero end
        if not State.flyLinearVel or not State.flyLinearVel.Parent then CreateLinearVelocityFly() end
        if State.flyLinearVel then State.flyLinearVel.VectorVelocity = State.flyCurrentVel end
        if State.flyAlignOrient then State.flyAlignOrient.CFrame = camCF end

    elseif State.flyMode == "CFrame" or State.flyMode == "CFrameBypass" then
        local rate = (target.Magnitude > 0.1) and State.flyAccel or State.flyBrake
        local alpha = math_clamp(rate * dt, 0, 1)
        State.flyCurrentVel = State.flyCurrentVel:Lerp(target, alpha)
        if State.flyCurrentVel.Magnitude < 0.3 then State.flyCurrentVel = Vector3_zero end
        local newPos = State.hrp.Position + State.flyCurrentVel * dt
        if State.flyAltLock and GetTierConfig().allowAltitudeLock then
            newPos = Vector3_new(newPos.X, State.flyAltHeight, newPos.Z)
        end
        local baseLook = flatLook
        if State.flyCurrentVel.Magnitude > 5 then
            local horiz = Vector3_new(State.flyCurrentVel.X, 0, State.flyCurrentVel.Z)
            if horiz.Magnitude > 0.1 then baseLook = horiz.Unit end
        end
        State.hrp.CFrame = CFrame_new(newPos, newPos + baseLook)
        State.hrp.AssemblyLinearVelocity = Vector3_zero
        State.hrp.AssemblyAngularVelocity = Vector3_zero

    elseif State.flyMode == "Behind" then
        local nearest, minDist = nil, math_huge
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local d = (hrp.Position - State.hrp.Position).Magnitude
                    if d < minDist then minDist = d; nearest = hrp end
                end
            end
        end
        if nearest then
            local behindPos = nearest.Position - nearest.CFrame.LookVector * 8 + Vector3_new(0, 3, 0)
            local dir = behindPos - State.hrp.Position
            if dir.Magnitude > 0.5 then target = dir.Unit * spd end
        end
        local rate = (target.Magnitude > 0.1) and State.flyAccel or State.flyBrake
        local alpha = math_clamp(rate * dt, 0, 1)
        State.flyCurrentVel = State.flyCurrentVel:Lerp(target, alpha)
        if State.flyCurrentVel.Magnitude < 0.3 then State.flyCurrentVel = Vector3_zero end
        if not State.flyBodyVel or not State.flyBodyVel.Parent then CreateBodyVelocityFly() end
        if State.flyBodyVel then State.flyBodyVel.Velocity = State.flyCurrentVel end
        if State.flyBodyGyro then State.flyBodyGyro.CFrame = camCF end

    elseif State.flyMode == "Tween" then
        if target.Magnitude > 0.1 then
            local newPos = State.hrp.Position + target.Unit * math_min(spd * dt, spd * 0.1)
            State.hrp.CFrame = CFrame_new(newPos, newPos + flatLook)
            State.hrp.AssemblyLinearVelocity = Vector3_zero
        end
    end
end)

        ResetOnSpawn = false, IgnoreGuiInset = true,
        DisplayOrder = 99995,
    })
    SafeParent(VJoyGui)
    VJoyGui.Enabled = false

    VJoyBase = mk("Frame", VJoyGui, {
        Size = UDim2_new(0, 60, 0, 140),
        Position = UDim2_new(1, -80, 1, -170),
        BackgroundColor3 = Theme.bg, BackgroundTransparency = 0.4,
        BorderSizePixel = 0, ZIndex = 100,
    })
    corner(VJoyBase, 30)
    stroke(VJoyBase, Theme.accent, 2, 0.3)

    VJoyThumb = mk("Frame", VJoyBase, {
        Size = UDim2_new(0, 44, 0, 44),
        Position = UDim2_new(0.5, -22, 0.5, -22),
        BackgroundColor3 = Theme.accent, BackgroundTransparency = 0.2,
        BorderSizePixel = 0, ZIndex = 101,
    })
    corner(VJoyThumb, 999)
    stroke(VJoyThumb, Color3_new(1, 1, 1), 2, 0.4)
    gradient(VJoyThumb, Theme.accent, Theme.accent2, 45)

    mk("TextLabel", VJoyThumb, {
        Size = UDim2_new(1, 0, 1, 0), BackgroundTransparency = 1,
        Text = "↕", TextColor3 = Color3_new(1, 1, 1),
        TextSize = 20, Font = Enum.Font.GothamBold, ZIndex = 102,
    })
    mk("TextLabel", VJoyBase, {
        Size = UDim2_new(1, 0, 0, 12), Position = UDim2_new(0, 0, 0, -14),
        BackgroundTransparency = 1, Text = "▲",
        TextColor3 = Color3_RGB(80, 255, 120),
        TextSize = 12, Font = Enum.Font.GothamBold, ZIndex = 102,
    })
    mk("TextLabel", VJoyBase, {
        Size = UDim2_new(1, 0, 0, 12), Position = UDim2_new(0, 0, 1, 2),
        BackgroundTransparency = 1, Text = "▼",
        TextColor3 = Color3_RGB(255, 100, 100),
        TextSize = 12, Font = Enum.Font.GothamBold, ZIndex = 102,
    })

    local function UpdateVJoy(input)
        if not VJoyBase.AbsoluteSize.Y or VJoyBase.AbsoluteSize.Y <= 0 then return end
        local center = VJoyBase.AbsolutePosition + VJoyBase.AbsoluteSize / 2
        local diffY = input.Position.Y - center.Y
        local maxDist = (VJoyBase.AbsoluteSize.Y / 2) - 22
        local clamped = math_clamp(diffY, -maxDist, maxDist)
        State.vjoyValue = -clamped / maxDist
        VJoyThumb.Position = UDim2_new(0.5, -22, 0.5, clamped - 22)
    end

    local function ResetVJoy()
        State.vjoyValue = 0
        State.vjoyActive = false
        if VJoyThumb then VJoyThumb.Position = UDim2_new(0.5, -22, 0.5, -22) end
    end

    VJoyBase.InputBegan:Connect(function(input)
        if input.UserInputType == UIT.Touch or input.UserInputType == UIT.MouseButton1 then
            VJoyTouch = input; State.vjoyActive = true; UpdateVJoy(input)
        end
    end)
    VJoyBase.InputEnded:Connect(function(input)
        if input == VJoyTouch then ResetVJoy(); VJoyTouch = nil end
    end)
    VJoyBase.InputChanged:Connect(function(input)
        if input == VJoyTouch then UpdateVJoy(input) end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == VJoyTouch then UpdateVJoy(input) end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input == VJoyTouch then ResetVJoy(); VJoyTouch = nil end
    end)
end

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if UserInputService:GetFocusedTextBox() then return end

    if input.KeyCode == KS.RightShift then
        if not KeyState.verified then
            SafeNotify("⚠️ Chưa xác thực", "Nhập key trước", 2); return
        end
        if Window and Window.Parent then
            if State.uiOpen then
                if closeMenu then closeMenu() end
            else
                if openMenu then openMenu() end
            end
        else
            print("[HOTKEY] Window nil, rebuilding...")
            BuildUI()
            task_wait(0.1)
            if openMenu then openMenu() end
        end
    elseif input.KeyCode == KS.P then
        if KeyState.verified then ToggleFly() end
    elseif input.KeyCode == KS.K then
        if KeyState.verified then
            if closeMenu then closeMenu() end
            if State.flyEnabled then StopFly() end
            if VJoyGui then VJoyGui.Enabled = false end
            if HJoyGui then HJoyGui.Enabled = false end
            SafeNotify("🚨 Panic", "Đã ẩn menu & tắt features", 2)
        end
    elseif input.KeyCode == KS.Space then
        if State.flyEnabled then
            State.vjoyActive = true
            State.vjoyValue = 1
        end
    elseif input.KeyCode == KS.LeftShift then
        if State.flyEnabled then
            State.vjoyActive = true
            State.vjoyValue = -1
        end
    end
end)

UserInputService.InputEnded:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == KS.Space or input.KeyCode == KS.LeftShift then
        if State.flyEnabled then
            State.vjoyActive = false
            State.vjoyValue = 0
        end
    end
end)

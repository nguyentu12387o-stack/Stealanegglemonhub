-- ═══════════ UI HELPERS ═══════════
local function mk(cls, parent, props)
    local i = New(cls)
    if props then for k, v in pairs(props) do pcall(function() i[k] = v end) end end
    if parent then pcall(function() i.Parent = parent end) end
    return i
end

local function corner(p, r)
    local c = New("UICorner")
    c.CornerRadius = UD(0, r or 6)
    c.Parent = p
end

local function stroke(p, col, thick, trans)
    local s = New("UIStroke")
    s.Color = col or C3RGB(0, 220, 255)
    s.Thickness = thick or 1
    s.Transparency = trans or 0.4
    s.Parent = p
    return s
end

local function gradient(p, c1, c2, rot)
    local g = New("UIGradient")
    g.Color = ColorSequence.new(c1, c2)
    g.Rotation = rot or 0
    g.Parent = p
end

local function listLay(p, pad)
    local l = New("UIListLayout")
    l.Padding = UD(0, pad or 4)
    l.SortOrder = Enum.SortOrder.LayoutOrder
    l.Parent = p
end

local function gridLay(p, cols, w, h, px, py)
    local g = New("UIGridLayout")
    g.CellSize = UD2(0, w or 100, 0, h or 26)
    g.CellPadding = UD2(0, px or 4, 0, py or 4)
    g.SortOrder = Enum.SortOrder.LayoutOrder
    g.FillDirection = Enum.FillDirection.Horizontal
    g.FillDirectionMaxCells = cols or 2
    g.Parent = p
end

local function uiPad(p, t, l, r, b)

    x.PaddingLeft = UD(0, l or 0)
    x.PaddingRight = UD(0, r or 0)

end

local flyNoclipConn = nil

local function DestroyFlyParts()
    if S.flyBV then pcall(function() S.flyBV:Destroy() end); S.flyBV = nil end
    if S.flyBG then pcall(function() S.flyBG:Destroy() end); S.flyBG = nil end
end

local function CreateBVFly()
    if not S.hrp then return end
    DestroyFlyParts()
    S.flyBV = mk("BodyVelocity", S.hrp, { MaxForce = V3(1e6, 1e6, 1e6), P = 1e5, Velocity = V3z })
    S.flyBG = mk("BodyGyro", S.hrp, { MaxTorque = V3(1e6, 1e6, 1e6), P = 1e4, CFrame = S.hrp.CFrame })
end

local function StartFly()
    UpdateChar()
    if not S.hrp or not S.hum then Notify("error", "Fly", "No char", 2); return end
    S.fly = true
    EnableFlyNoclip()
    S.flyVel = V3z
    if S.flyMode == "BodyVelocity" or S.flyMode == "Camera" then
        CreateBVFly()
        pcall(function() S.hum.PlatformStand = true end)
    end
    if VJGui then VJGui.Enabled = true end
    Notify("success", "✈️ FLY ON", S.flyMode, 2)
end

local function StopFly()
    S.fly = false
    DisableFlyNoclip()
    DestroyFlyParts()
    pcall(function() if S.hum then S.hum.PlatformStand = false end end)
    if VJGui then VJGui.Enabled = false end
    S.vjoy = 0
    Notify("warn", "✈️ FLY OFF", "", 2)
end

local function ToggleFly() if S.fly then StopFly() else StartFly() end end

RunService.RenderStepped:Connect(function(dt)
    if not S.fly or not S.hrp or not S.hum then return end
    if S.hum.Health <= 0 then StopFly(); return end
    dt = m_min(dt, 0.05)

    local camCF = Camera.CFrame
    local look = camCF.LookVector
    local right = camCF.RightVector
    local fL = V3(look.X, 0, look.Z)
    local fR = V3(right.X, 0, right.Z)
    if fL.Magnitude > 0.01 then fL = fL.Unit end
    if fR.Magnitude > 0.01 then fR = fR.Unit end

    local spd = m_clamp(S.flySpd, 1, GetTier().maxSpd)
    local target = V3z

    if S.vjoyAct and m_abs(S.vjoy) > 0.05 then
        target = target + V3(0, 1, 0) * (S.vjoy * spd)
    end

    local move = S.hum.MoveDirection
    if move.Magnitude > 0.1 then
        target = target + (fL * move.Z + fR * move.X) * spd
    end

    local rate = (target.Magnitude > 0.1) and S.flyAccel or S.flyBrake
    local alpha = m_clamp(rate * dt, 0, 1)
    S.flyVel = S.flyVel:Lerp(target, alpha)
    if S.flyVel.Magnitude < 0.3 then S.flyVel = V3z end

    if S.flyMode == "BodyVelocity" or S.flyMode == "Camera" then
        if not S.flyBV or not S.flyBV.Parent then CreateBVFly() end
        if S.flyBV then S.flyBV.Velocity = S.flyVel end
        if S.flyBG then S.flyBG.CFrame = camCF end
        if S.flyMode == "Camera" then
            S.hrp.CFrame = CF(S.hrp.Position, S.hrp.Position + look)
        end
    else
        local newPos = S.hrp.Position + S.flyVel * dt
        local bL = fL
        if S.flyVel.Magnitude > 5 then
            local hz = V3(S.flyVel.X, 0, S.flyVel.Z)
            if hz.Magnitude > 0.1 then bL = hz.Unit end
        end
        S.hrp.CFrame = CF(newPos, newPos + bL)
        S.hrp.AssemblyLinearVelocity = V3z
        S.hrp.AssemblyAngularVelocity = V3z
    end
end)

local VJGui, VJBase, VJThumb, VJTouch

local function CreateVJoy()
    if VJGui then return end
    VJGui = mk("ScreenGui", nil, {
        Name = "BlackZVJoy", ResetOnSpawn = false,
        IgnoreGuiInset = true, DisplayOrder = 99995,
    })
    SafeParent(VJGui)
    VJGui.Enabled = false

    VJBase = mk("Frame", VJGui, {
        Size = UD2(0, 60, 0, 140),
        Position = UD2(1, -80, 1, -170),
        BackgroundColor3 = Theme.bg, BackgroundTransparency = 0.4,
        BorderSizePixel = 0, ZIndex = 100,
    })
    corner(VJBase, 30)
    stroke(VJBase, Theme.accent, 2, 0.3)

    VJThumb = mk("Frame", VJBase, {
        Size = UD2(0, 44, 0, 44),
        Position = UD2(0.5, -22, 0.5, -22),
        BackgroundColor3 = Theme.accent, BackgroundTransparency = 0.2,
        BorderSizePixel = 0, ZIndex = 101,
    })
    corner(VJThumb, 999)
    mk("TextLabel", VJThumb, {
        Size = UD2(1, 0, 1, 0), BackgroundTransparency = 1,
        Text = "↕", TextColor3 = C3RGB(255, 255, 255),
        TextSize = 20, Font = Enum.Font.GothamBold, ZIndex = 102,
    })

    local function upd(inp)
        if not VJBase.AbsoluteSize.Y or VJBase.AbsoluteSize.Y <= 0 then return end
        local c = VJBase.AbsolutePosition + VJBase.AbsoluteSize / 2
        local dY = inp.Position.Y - c.Y
        local m = (VJBase.AbsoluteSize.Y / 2) - 22
        local cl = m_clamp(dY, -m, m)
        S.vjoy = -cl / m
        VJThumb.Position = UD2(0.5, -22, 0.5, cl - 22)
    end

    local function rst()
        S.vjoy = 0; S.vjoyAct = false
        if VJThumb then VJThumb.Position = UD2(0.5, -22, 0.5, -22) end
    end

    VJBase.InputBegan:Connect(function(i)
        if i.UserInputType == UIT.Touch or i.UserInputType == UIT.MouseButton1 then
            VJTouch = i; S.vjoyAct = true; upd(i)
        end
    end)
    VJBase.InputEnded:Connect(function(i) if i == VJTouch then rst(); VJTouch = nil end end)
    UIS.InputChanged:Connect(function(i) if i == VJTouch then upd(i) end end)
    UIS.InputEnded:Connect(function(i) if i == VJTouch then rst(); VJTouch = nil end end)
end

            BackgroundColor3 = Theme.bgTrk, BorderSizePixel = 0, ZIndex = 12,
        })
        corner(tr, 3)

            Size = UD2(m_clamp((dv - mn) / (mx - mn), 0, 1), 0, 1, 0),
            BackgroundColor3 = Theme.accent, BorderSizePixel = 0, ZIndex = 13,
        })
        corner(fl, 3)

        local dr = false
        local function upd(i)
            if tr.AbsoluteSize.X <= 0 then return end
            local r = m_clamp((i.Position.X - tr.AbsolutePosition.X) / tr.AbsoluteSize.X, 0, 1)
            fl.Size = UD2(r, 0, 1, 0)
            local v = m_floor(mn + (mx - mn) * r)
            vl.Text = tostring(v)
            pcall(cb, v)
        end
        tr.InputBegan:Connect(function(i)
            if i.UserInputType == UIT.MouseButton1 or i.UserInputType == UIT.Touch then
                dr = true; upd(i)
            end
        end)
        tr.InputEnded:Connect(function() dr = false end)
        UIS.InputChanged:Connect(function(i)
            if dr and (i.UserInputType == UIT.MouseMovement or i.UserInputType == UIT.Touch) then
                upd(i)
            end
        end)
    end

    local drag, ds, sp
    b.InputBegan:Connect(function(i)
        if i.UserInputType == UIT.MouseButton1 or i.UserInputType == UIT.Touch then
            drag = true; ds = i.Position; sp = b.Position
        end
    end)
    b.InputEnded:Connect(function() drag = false end)
    UIS.InputChanged:Connect(function(i)
        if drag and (i.UserInputType == UIT.MouseMovement or i.UserInputType == UIT.Touch) then
            local d = i.Position - ds
            b.Position = UD2(sp.X.Scale, sp.X.Offset + d.X, sp.Y.Scale, sp.Y.Offset + d.Y)
        end
    end)
    return b
end

-- ═══════════ HOTKEYS ═══════════
UIS.InputBegan:Connect(function(i, gp)
    if gp then return end
    if UIS:GetFocusedTextBox() then return end

    if i.KeyCode == KS.RightShift then
        if not KeyState.verified then return end
        if S.menuOpen then
            if closeMenu then closeMenu() end
        else
            if openMenu then openMenu() end
        end
    elseif i.KeyCode == KS.P then
        if KeyState.verified then ToggleFly() end
    elseif i.KeyCode == KS.K then
        if KeyState.verified then
            if closeMenu then closeMenu() end
            if S.fly then StopFly() end
            if VJGui then VJGui.Enabled = false end
        end
    elseif i.KeyCode == KS.Space then
        if S.fly then S.vjoyAct = true; S.vjoy = 1 end
    elseif i.KeyCode == KS.LeftShift then
        if S.fly then S.vjoyAct = true; S.vjoy = -1 end
    end
end)

UIS.InputEnded:Connect(function(i, gp)
    if gp then return end
    if i.KeyCode == KS.Space or i.KeyCode == KS.LeftShift then
        if S.fly then S.vjoyAct = false; S.vjoy = 0 end
    end
end)

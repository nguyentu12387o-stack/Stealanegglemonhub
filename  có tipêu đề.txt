local function CleanupOldGuis()
    local allParents = { CoreGui }
    
    pcall(function()
        if LP:FindFirstChild("PlayerGui") then
            table_insert(allParents, LP.PlayerGui)
        end
    end)
    
    pcall(function()
        if gethui then
            local ok, h = pcall(gethui)
            if ok and h then table_insert(allParents, h) end
        end
    end)
    
    local cleanedCount = 0
    for _, parent in ipairs(allParents) do
        if parent then
            for _, child in ipairs(parent:GetChildren()) do
                if child:IsA("ScreenGui") then
                    local n = child.Name
                    if string_find(n, "BlackZ") or string_find(n, "System") 
                        or string_find(n, "Rezzy") or string_find(n, "Key") then
                        pcall(function()
                            child.Enabled = false
                            child:Destroy()
                        end)
                        cleanedCount = cleanedCount + 1
                    end
                end
            end
        end
    end
    
    -- Xóa ESP folder cũ
    pcall(function()
        for _, obj in ipairs(Workspace:GetChildren()) do
            if obj.Name == "BlackZESP" or string_find(obj.Name, "BlackZESP_") then
                obj:Destroy()
                cleanedCount = cleanedCount + 1
            end
        end
    end)
    
    _G.BlackZ_Built = false
    _G.BlackZ_Alive = true
    _G.BlackZ_Version = "v8.6"
    
    print("[CLEANUP] Đã xóa " .. cleanedCount .. " GUI cũ")
end

CleanupOldGuis()

local function VerifyKey(inputKey)
    if not inputKey or inputKey == "" then
        return false, "Vui lòng nhập key"
    end
    inputKey = string_lower(inputKey):gsub("^%s+", ""):gsub("%s+$", "")
    local db = KeyDatabase[inputKey]
    if not db then
        return false, "Key không tồn tại"
    end
    KeyState.verified = true
    KeyState.tier = db.tier
    KeyState.key = inputKey
    return true, "Xác thực thành công"
end

local function GetTierConfig()
    return TierConfig[KeyState.tier] or TierConfig.FREE

local function mk(cls, parent, props)
    local i = Instance_new(cls)
    if props then
        for k, v in pairs(props) do
            pcall(function() i[k] = v end)
        end
    end
    if parent then pcall(function() i.Parent = parent end) end
    return i
end

local function corner(p, r)
    local c = Instance_new("UICorner")
    pcall(function() c.CornerRadius = UDim_new(0, r or 6); c.Parent = p end)
    return c
end

local function stroke(p, col, thick, trans)
    local s = Instance_new("UIStroke")
    pcall(function()
        s.Color = col or Color3_new(1, 1, 1)
        s.Thickness = thick or 1
        s.Transparency = trans or 0
        s.Parent = p
    end)
    return s
end

local function gradient(p, c1, c2, rot)
    local g = Instance_new("UIGradient")
    pcall(function()
        g.Color = ColorSequence.new(c1, c2)
        g.Rotation = rot or 0
        g.Parent = p
    end)
    return g
end

local function listLayout(p, pad)
    local l = Instance_new("UIListLayout")
    pcall(function()
        l.Padding = UDim_new(0, pad or 4)
        l.SortOrder = Enum.SortOrder.LayoutOrder
        l.Parent = p
    end)
    return l
end

local function gridLayout(p, cols, cellW, cellH, padX, padY)
    local g = Instance_new("UIGridLayout")
    pcall(function()
        g.CellSize = UDim2_new(0, cellW or 100, 0, cellH or 26)
        g.CellPadding = UDim2_new(0, padX or 4, 0, padY or 4)
        g.SortOrder = Enum.SortOrder.LayoutOrder
        g.FillDirection = Enum.FillDirection.Horizontal
        g.FillDirectionMaxCells = cols or 2
        g.Parent = p
    end)
    return g
end

local function uiPadding(p, t, l, r, b)
    local x = Instance_new("UIPadding")
    pcall(function()
        x.PaddingTop = UDim_new(0, t or 0)
        x.PaddingLeft = UDim_new(0, l or 0)
        x.PaddingRight = UDim_new(0, r or 0)
        x.PaddingBottom = UDim_new(0, b or 0)
        x.Parent = p
    end)
    return x
end

local function tween(obj, time, props)

local function UpdateChar()
    local c = LP.Character
    if c ~= State.char then
        State.char = c
        if c then
            State.hrp = c:FindFirstChild("HumanoidRootPart")
            State.hum = c:FindFirstChildOfClass("Humanoid")
        else
            State.hrp = nil
            State.hum = nil
        end
    else
        if State.char then
            State.hrp = State.char:FindFirstChild("HumanoidRootPart")
            State.hum = State.char:FindFirstChildOfClass("Humanoid")
        end
    end
end

local invisibleConn = nil

ApplyInvisible = function(on)
    if not State.char then
        UpdateChar()
        if not State.char then return end
    end
    pcall(function()
        for _, v in ipairs(State.char:GetDescendants()) do
            if v:IsA("BasePart") then

                v.LocalTransparencyModifier = on and 1 or 0
                v.CastShadow = not on
            elseif v:IsA("Decal") or v:IsA("Texture") then
                v.Transparency = on and 1 or 0
            elseif v:IsA("Accessory") then
                local handle = v:FindFirstChild("Handle")

                State.hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.Viewer

        end
    end)
end

            pcall(function() obj:Destroy() end)
            State.espObjects[key] = nil
        end
    end

        end

    if State.flyBodyGyro then pcall(function() State.flyBodyGyro:Destroy() end); State.flyBodyGyro = nil end

    elseif State.flyMode == "Camera" then

StartFly = function()
    if not State.hrp or not State.hum then
        UpdateChar()
        if not State.hrp or not State.hum then
            Notify("error", "Fly", "Không tìm thấy nhân vật", 2)
            return
        end
    end
    State.flyEnabled = true
    EnableFlyNoclip()
    State.flyCurrentVel = Vector3_zero
    State.flyTargetEntity = nil
    
    CreateFlyPartsForMode()
    
    if State.flyMode == "BodyVelocity" or State.flyMode == "LinearVelocity" or State.flyMode == "Camera" then
        pcall(function()
            State.hum.PlatformStand = true
            State.hum:SetStateEnabled(HS.Ragdoll, false)
            State.hum:SetStateEnabled(HS.FallingDown, false)
        end)
    end
    
    if State.vjoyEnabled and VJoyGui then
        VJoyGui.Enabled = true
    end
    if State.hjoyEnabled and HJoyGui then
        HJoyGui.Enabled = true
    end
    
    Notify("success", "✈️ FLY ON", "Mode: " .. State.flyMode, 2)
end

StopFly = function()
    State.flyEnabled = false
    DisableFlyNoclip()
    DestroyFlyParts()
    pcall(function()
        if State.hum then
            State.hum.PlatformStand = false
        end
    end)
    if VJoyGui then VJoyGui.Enabled = false end
    if HJoyGui then HJoyGui.Enabled = true end
    State.vjoyValue = 0
    State.hjoyValue = Vector2_zero
    State.flyTargetEntity = nil
    Notify("warn", "✈️ FLY OFF", "", 2)
end

ToggleFly = function()
    if State.flyEnabled then
        StopFly()
    else
        StartFly()
    end
end

-- ═══ FLY RENDER LOOP — 5 MODES ═══
RunService.RenderStepped:Connect(function(dt)
    if not State.flyEnabled then return end
    if not State.hrp or not State.hum then return end
    if State.hum.Health <= 0 then StopFly() return end

    dt = math_min(dt, 0.05)
    local camCF = Camera.CFrame
    local lookVec = camCF.LookVector
    local rightVec = camCF.RightVector

    local flatLook = Vector3_new(lookVec.X, 0, lookVec.Z)
    local flatRight = Vector3_new(rightVec.X, 0, rightVec.Z)
    if flatLook.Magnitude > 0.01 then flatLook = flatLook.Unit end
    if flatRight.Magnitude > 0.01 then flatRight = flatRight.Unit end

    local spd = math_clamp(State.flySpeed, 1, GetTierConfig().maxFlySpeed)
    local target = Vector3_zero

    -- Vertical input (V-Joystick + Space/LShift)
    if State.vjoyActive and math_abs(State.vjoyValue) > 0.05 then
        target = target + Vector3_new(0, 1, 0) * (State.vjoyValue * spd)
    end

    -- Horizontal input (WASD / MoveDirection)
    local move = State.hum.MoveDirection
    if move.Magnitude > 0.1 then
        target = target + (flatLook * move.Z + flatRight * move.X) * spd
    end

    -- Horizontal input (H-Joystick)
    if State.hjoyActive and State.hjoyValue.Magnitude > 0.1 then
        local dir = (flatRight * State.hjoyValue.X + flatLook * -State.hjoyValue.Y)
        if dir.Magnitude > 0.01 then
            target = target + dir.Unit * spd
        end
    end

    -- ═══ Mode 1: CFrameBypass ═══
    if State.flyMode == "CFrameBypass" or State.flyMode == "CFrame" then
        local rate = (target.Magnitude > 0.1) and State.flyAccel or State.flyBrake
        local alpha = math_clamp(rate * dt, 0, 1)
        State.flyCurrentVel = State.flyCurrentVel:Lerp(target, alpha)
        if State.flyCurrentVel.Magnitude < 0.3 then
            State.flyCurrentVel = Vector3_zero
        end
        
        local newPos = State.hrp.Position + State.flyCurrentVel * dt
        if State.flyAltLock and GetTierConfig().allowAltitudeLock then
            newPos = Vector3_new(newPos.X, State.flyAltHeight, newPos.Z)
        end
        
        local baseLook = flatLook
        if State.flyCurrentVel.Magnitude > 5 then
            local horiz = Vector3_new(State.flyCurrentVel.X, 0, State.flyCurrentVel.Z)
            if horiz.Magnitude > 0.1 then baseLook = horiz.Unit end
        end
        State.hrp.CFrame = CFrame_new(newPos, newPos + baseLook)
        State.hrp.AssemblyLinearVelocity = Vector3_zero
        State.hrp.AssemblyAngularVelocity = Vector3_zero

    -- ═══ Mode 2: Camera ═══
    elseif State.flyMode == "Camera" then
        local rate = (target.Magnitude > 0.1) and State.flyAccel or State.flyBrake
        local alpha = math_clamp(rate * dt, 0, 1)
        State.flyCurrentVel = State.flyCurrentVel:Lerp(target, alpha)
        if State.flyCurrentVel.Magnitude < 0.3 then
            State.flyCurrentVel = Vector3_zero
        end
        
        local newPos = State.hrp.Position + State.flyCurrentVel * dt
        State.hrp.CFrame = CFrame_new(newPos, newPos + lookVec)
        State.hrp.AssemblyLinearVelocity = Vector3_zero
        State.hrp.AssemblyAngularVelocity = Vector3_zero
        if State.flyBodyGyro then
            State.flyBodyGyro.CFrame = camCF
        end

    -- ═══ Mode 3: BodyVelocity ═══
    elseif State.flyMode == "BodyVelocity" then
        local rate = (target.Magnitude > 0.1) and State.flyAccel or State.flyBrake
        local alpha = math_clamp(rate * dt, 0, 1)
        State.flyCurrentVel = State.flyCurrentVel:Lerp(target, alpha)
        if State.flyCurrentVel.Magnitude < 0.3 then
            State.flyCurrentVel = Vector3_zero
        end
        
        if not State.flyBodyVel or not State.flyBodyVel.Parent then
            CreateBodyVelocityFly()
        end
        if State.flyBodyVel then
            State.flyBodyVel.Velocity = State.flyCurrentVel
        end
        if State.flyBodyGyro then
            State.flyBodyGyro.CFrame = camCF
        end

    -- ═══ Mode 4: LinearVelocity ═══
    elseif State.flyMode == "LinearVelocity" then
        local rate = (target.Magnitude > 0.1) and State.flyAccel or State.flyBrake
        local alpha = math_clamp(rate * dt, 0, 1)
        State.flyCurrentVel = State.flyCurrentVel:Lerp(target, alpha)
        if State.flyCurrentVel.Magnitude < 0.3 then
            State.flyCurrentVel = Vector3_zero
        end
        
        if not State.flyLinearVel or not State.flyLinearVel.Parent then
            CreateLinearVelocityFly()
        end
        if State.flyLinearVel then
            State.flyLinearVel.VectorVelocity = State.flyCurrentVel
        end
        if State.flyAlignOrient then
            State.flyAlignOrient.CFrame = camCF
        end

    -- ═══ Mode 5: Behind ═══
    elseif State.flyMode == "Behind" then
        local nearest, minDist = nil, math_huge
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local d = (hrp.Position - State.hrp.Position).Magnitude
                    if d < minDist then
                        minDist = d
                        nearest = hrp
                    end
                end
            end
        end
        if nearest then
            State.flyTargetEntity = nearest
            local behindPos = nearest.Position - nearest.CFrame.LookVector * 8 + Vector3_new(0, 3, 0)
            local dir = behindPos - State.hrp.Position
            if dir.Magnitude > 0.5 then
                target = dir.Unit * spd
            end
        end
        local rate = (target.Magnitude > 0.1) and State.flyAccel or State.flyBrake
        local alpha = math_clamp(rate * dt, 0, 1)
        State.flyCurrentVel = State.flyCurrentVel:Lerp(target, alpha)
        if State.flyCurrentVel.Magnitude < 0.3 then
            State.flyCurrentVel = Vector3_zero
        end
        
        local newPos = State.hrp.Position + State.flyCurrentVel * dt
        State.hrp.CFrame = CFrame_new(newPos, newPos + flatLook)
        State.hrp.AssemblyLinearVelocity = Vector3_zero
        State.hrp.AssemblyAngularVelocity = Vector3_zero
    end
end)

    HJoyBase.InputChanged:Connect(function(input)
        if input == HJoyTouch then
            UpdateHJoy(input)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == HJoyTouch then
            UpdateHJoy(input)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input == HJoyTouch then
            ResetHJoy()
            HJoyTouch = nil
        end
    end)
end

LF.o.fogStart = Lighting.FogStart

-- 1. Low Graphics
local function SetLowGraphics(on)
    pcall(function()
        if on then
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
            Lighting.Brightness = 1
            Lighting.GlobalShadows = false
            Lighting.EnvironmentDiffuseScale = 0
            Lighting.EnvironmentSpecularScale = 0
        else
            Lighting.Brightness = LF.o.brightness
            Lighting.GlobalShadows = LF.o.shadows
        end
    end)
end

-- 2. Remove Fog
local function RemoveFog(on)
    pcall(function()
        if on then
            Lighting.FogEnd = 1e9
            Lighting.FogStart = 1e9
            Lighting.FogColor = Color3_RGB(0, 0, 0)
        else
            Lighting.FogEnd = LF.o.fogEnd
            Lighting.FogStart = LF.o.fogStart
        end
    end)
end

-- 3. No Ambient
local function NoAmbient(on)
    pcall(function()
        if on then
            Lighting.Ambient = Color3_RGB(0, 0, 0)
            Lighting.OutdoorAmbient = Color3_RGB(0, 0, 0)
            Lighting.EnvironmentDiffuseScale = 0
            Lighting.EnvironmentSpecularScale = 0
        else
            Lighting.Ambient = LF.o.ambient
            Lighting.OutdoorAmbient = LF.o.outdoor
        end
    end)
end

-- 4. Reduce Shadows
local function ReduceShadows(on)
    pcall(function()
        if on then
            Lighting.GlobalShadows = false
        else
            Lighting.GlobalShadows = LF.o.shadows
        end
    end)
end

-- 5. Remove Textures
local function RemoveTextures(on)
    pcall(function()
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("Texture") then
                if on then
                    if not obj:GetAttribute("BZ_T") then
                        obj:SetAttribute("BZ_T", obj.Transparency)
                        obj.Transparency = 1
                    end
                else
                    local o = obj:GetAttribute("BZ_T")
                    if o then
                        obj.Transparency = o
                        obj:SetAttribute("BZ_T", nil)
                    end
                end
            end
        end
    end)
end

-- 6. Remove Decals
local function RemoveDecals(on)
    pcall(function()
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("Decal") then
                if on then
                    if not obj:GetAttribute("BZ_D") then
                        obj:SetAttribute("BZ_D", obj.Transparency)
                        obj.Transparency = 1
                    end
                else
                    local o = obj:GetAttribute("BZ_D")
                    if o then
                        obj.Transparency = o
                        obj:SetAttribute("BZ_D", nil)
                    end
                end
            end
        end
    end)
end

-- 7. Remove Particles
local function RemoveParticles(on)
    pcall(function()
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("ParticleEmitter") or obj:IsA("Trail")
                or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles") then
                if on then
                    if obj.Enabled then
                        obj:SetAttribute("BZ_P", true)
                        obj.Enabled = false
                    end
                else
                    if obj:GetAttribute("BZ_P") then
                        obj.Enabled = true
                        obj:SetAttribute("BZ_P", nil)
                    end
                end
            end
        end
    end)
end

-- 8. Disable Skybox
local function DisableSkybox(on)
    pcall(function()
        if on then
            local sky = Lighting:FindFirstChildOfClass("Sky")
            if sky then
                _G.BZ_Sky = sky
                sky.Parent = nil
            end
        else
            if _G.BZ_Sky and not _G.BZ_Sky.Parent then
                _G.BZ_Sky.Parent = Lighting
                _G.BZ_Sky = nil
            end
        end
    end)
end

-- 9. Destroy Terrain Effects
local function DestroyTerrainFX(on)
    pcall(function()
        local terrain = Workspace:FindFirstChildOfClass("Terrain")
        if terrain and on then
            terrain.WaterWaveSize = 0
            terrain.WaterWaveSpeed = 0
            terrain.WaterReflectance = 0
            terrain.WaterTransparency = 1
            terrain.Decoration = false
        end
    end)
end

-- 10. Disable Sounds
local function DisableSounds(on)
    pcall(function()
        if on then
            SoundService.AmbientReverb = Enum.ReverbType.NoReverb
        end
    end)
end

-- 11. Remove Lighting Effects
local function RemoveLightingFX(on)
    pcall(function()
        for _, obj in ipairs(Lighting:GetChildren()) do
            if obj:IsA("BlurEffect") or obj:IsA("SunRaysEffect")
                or obj:IsA("ColorCorrectionEffect") or obj:IsA("BloomEffect")
                or obj:IsA("DepthOfFieldEffect") then
                if on then
                    obj:SetAttribute("BZ_E", obj.Enabled)
                    obj.Enabled = false
                else
                    local o = obj:GetAttribute("BZ_E")
                    if o ~= nil then
                        obj.Enabled = o
                        obj:SetAttribute("BZ_E", nil)
                    end
                end
            end
        end
    end)
end

function UI.slider(parent, text, minV, maxV, defaultV, callback)
    local c = mk("Frame", parent, {
        Size = UDim2_new(1, 0, 0, 38),
        BackgroundColor3 = Theme.bgCard,
        BackgroundTransparency = 0.25,
        BorderSizePixel = 0,
        ZIndex = 11,

    corner(c, 6)

    mk("TextLabel", c, {
        Size = UDim2_new(1, -60, 0, 14),
        Position = UDim2_new(0, 8, 0, 3),
        BackgroundTransparency = 1,
        Text = text,
        TextColor3 = Theme.txt,
        Font = Enum.Font.GothamBold,
        TextSize = 10,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 12,
    })
    local valLbl = mk("TextLabel", c, {
        Size = UDim2_new(0, 50, 0, 14),
        Position = UDim2_new(1, -58, 0, 3),
        BackgroundTransparency = 1,
        Text = tostring(defaultV),
        Font = Enum.Font.GothamBlack,
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Right,
        ZIndex = 12,
    })
    gradient(valLbl, Theme.accent, Theme.accent2, 0)

    local track = mk("Frame", c, {
        Size = UDim2_new(1, -16, 0, 5),
        Position = UDim2_new(0, 8, 0, 24),
        BackgroundColor3 = Theme.bgTrk,
        BorderSizePixel = 0,
        ZIndex = 12,
    })
    corner(track, 3)

    local fill = mk("Frame", track, {
        Size = UDim2_new(math_clamp((defaultV - minV) / (maxV - minV), 0, 1), 0, 1, 0),
        BackgroundColor3 = Theme.accent,
        BorderSizePixel = 0,
        ZIndex = 13,
    })
    corner(fill, 3)
    gradient(fill, Theme.accent, Theme.accent2, 0)

    local dragging = false
    local function update(input)
        if track.AbsoluteSize.X <= 0 then return end
        local ratio = math_clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        fill.Size = UDim2_new(ratio, 0, 1, 0)
        local v = math_floor(minV + (maxV - minV) * ratio)
        valLbl.Text = tostring(v)
        pcall(callback, v)
    end

    track.InputBegan:Connect(function(input)
        if input.UserInputType == UIT.MouseButton1 or input.UserInputType == UIT.Touch then
            dragging = true
            update(input)
        end
    end)
    track.InputEnded:Connect(function() dragging = false end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == UIT.MouseMovement or input.UserInputType == UIT.Touch) then
            update(input)
        end
    end)
    return c
end

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if UserInputService:GetFocusedTextBox() then return end

    if input.KeyCode == KS.RightShift then
        if not KeyState.verified then
            SafeNotify("⚠️ Chưa xác thực", "Nhập key trước", 2)
            return
        end
        if Window and Window.Parent then
            if State.uiOpen then
                if closeMenu then closeMenu() end
            else
                if openMenu then openMenu() end
            end
        else
            BuildUI()
            task_wait(0.1)
            if openMenu then openMenu() end
        end
    elseif input.KeyCode == KS.P then
        if KeyState.verified then ToggleFly() end
    elseif input.KeyCode == KS.K then
        if KeyState.verified then
            if closeMenu then closeMenu() end
            if State.flyEnabled then StopFly() end
            if VJoyGui then VJoyGui.Enabled = false end
            if HJoyGui then HJoyGui.Enabled = false end
            SafeNotify("🚨 Panic", "Đã ẩn menu & tắt features", 2)
        end
    elseif input.KeyCode == KS.Space then
        if State.flyEnabled then
            State.vjoyActive = true
            State.vjoyValue = 1
        end
    elseif input.KeyCode == KS.LeftShift then
        if State.flyEnabled then
            State.vjoyActive = true
            State.vjoyValue = -1
        end
    end
end)

UserInputService.InputEnded:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == KS.Space or input.KeyCode == KS.LeftShift then
        if State.flyEnabled then
            State.vjoyActive = false
            State.vjoyValue = 0
        end
    end
end)

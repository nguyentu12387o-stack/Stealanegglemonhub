local Lifecycle = {
    connections = {},
    threads = {},
    destroyed = false,
}
function Lifecycle.TrackConnection(conn)
    if not conn then return nil end
    table_insert(Lifecycle.connections, conn)
    return conn
end
function Lifecycle.TrackThread(th)
    if not th then return nil end
    table_insert(Lifecycle.threads, th)
    return th
end
function Lifecycle.Destroy()
    if Lifecycle.destroyed then return end
    Lifecycle.destroyed = true
    for _, c in ipairs(Lifecycle.connections) do
        pcall(function() c:Disconnect() end)
    end
    Lifecycle.connections = {}
    Lifecycle.threads = {}
end

local function ContainsKeyword(str, keywordList)
    if type(str) ~= "string" then return false end
    local lower = string_lower(str)
    for i = 1, #keywordList do
        if string_find(lower, keywordList[i], 1, true) then
            return true
        end
    end
    return false
end

function Humanizer.TweenTeleport(hrp, targetPos, duration)

    local endCF
    pcall(function()
        endCF = CFrame_new(targetPos, targetPos + hrp.CFrame.LookVector)

            TweenInfo_new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            { CFrame = endCF }
        )
        tween:Play()
    end)
    return tween
end
function Humanizer.Jitter(pos, amount)
    amount = amount or 1.5
    return pos + Vector3_new(
        (math_random() - 0.5) * amount,
        (math_random() - 0.5) * amount,
        (math_random() - 0.5) * amount
    )
end

    local i = Instance_new(cls)

        for k, v in pairs(props) do
            pcall(function() i[k] = v end)
        end
    end

    return i
end

local RemoteFire = {}

local REMOTE_KEYWORDS = {
    "egg", "claim", "collect", "steal", "hatch",
    "open", "spawn", "get", "reward", "pickup",
    "pick", "obtain", "take", "grab", "buy", "purchase",
}

local function IsEggRemote(name)
    if type(name) ~= "string" then return false end
    local low = string_lower(name)
    for _, kw in ipairs(REMOTE_KEYWORDS) do
        if string_find(low, kw, 1, true) then return true end
    end
    return false
end

local function IsCustomRemote(name)
    if type(name) ~= "string" then return false end
    local low = string_lower(name)
    for _, custom in ipairs(State.customRemoteNames) do
        if string_lower(custom) == low then return true end
    end
    return false
end

local function RecursiveScan(root, results, seenObjs, depth)
    depth = depth or 0
    if depth > 8 or not root then return end

    local ok, children = pcall(function() return root:GetChildren() end)
    if not ok or not children then return end

    for _, child in ipairs(children) do
        if (child:IsA("RemoteEvent") or child:IsA("RemoteFunction"))
            and not seenObjs[child] then
            if IsEggRemote(child.Name) or IsCustomRemote(child.Name) then
                seenObjs[child] = true
                table_insert(results, {
                    obj = child,
                    name = child.Name,
                    type = child:IsA("RemoteEvent") and "RE" or "RF",
                })
            end
        elseif child:IsA("Folder") or child:IsA("Model") or child:IsA("Actor") then
            RecursiveScan(child, results, seenObjs, depth + 1)
        end
    end
end

function RemoteFire.Scan()
    State.scannedRemotes = {}
    local seenObjs = {}
    local results = State.scannedRemotes

    local pkg = ReplicatedStorage:FindFirstChild("Packages")
    if pkg then
        local net = pkg:FindFirstChild("Networking")
        if net then
            RecursiveScan(net, results, seenObjs, 0)
        end
    end

    local commonFolders = {
        "RE", "RF", "Remotes", "RemoteEvents", "Events", "Networking",
        "Network", "Communication", "API",
    }
    for _, fname in ipairs(commonFolders) do
        local folder = ReplicatedStorage:FindFirstChild(fname)
        if folder then
            RecursiveScan(folder, results, seenObjs, 0)
        end
    end

    RecursiveScan(ReplicatedStorage, results, seenObjs, 0)

    Logger.log("RemoteFire", string_format("Scanned %d remotes", #State.scannedRemotes))
    return #State.scannedRemotes
end

local LAST_SCAN_ATTEMPT = 0
local function EnsureScanned()
    if #State.scannedRemotes > 0 then return true end
    local now = os_clock()
    if now - LAST_SCAN_ATTEMPT < 5 then return false end
    LAST_SCAN_ATTEMPT = now
    RemoteFire.Scan()
    return #State.scannedRemotes > 0
end

function RemoteFire.FireAll()
    if not EnsureScanned() then return 0 end
    local count = 0
    for _, r in ipairs(State.scannedRemotes) do
        if r.obj and r.obj.Parent then
            if r.type == "RE" then
                local ok = pcall(function()
                    r.obj:FireServer()
                end)
                if ok then count = count + 1 end
            else
                task_spawn(function()
                    pcall(function() r.obj:InvokeServer() end)
                end)
                count = count + 1
            end
        end
    end
    return count
end

    else

            print(string_format("  [%d] %s (%s)", i, r.name, r.type))
        end
    end
    print("═══════════════════════════════════════")
end

    for k in pairs(espObjects) do
        table_insert(keys, k)

        DestroyESPForPlayer(k)
    end
    espObjects = {}

local function RefreshESP()
    if not State.esp then
        ClearAllESP()
        return
    end

    local stale = {}
    for player, data in pairs(espObjects) do
        if not player.Parent
            or not player.Character
            or (data.hl and data.hl.Adornee ~= player.Character) then
            table_insert(stale, player)
        end
    end
    for _, player in ipairs(stale) do
        DestroyESPForPlayer(player)
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            local data = espObjects[player]
            if not data then
                CreateESPForPlayer(player)
            else
                local col = Color3_fromHSV((os_clock() * 0.3) % 1, 1, 1)
                pcall(function() data.hl.FillColor = col end)
                pcall(function() data.lbl.TextColor3 = col end)

    EggCache.folderRefs = {}

    pcall(function()

            if string_find(string_lower(child.Name), "egg", 1, true) then
                table_insert(EggCache.folderRefs, child)
            end
        end
        if #EggCache.folderRefs == 0 then
            local common = { "Map", "Spawned", "Items", "Objects", "World", "Eggs" }
            for _, fname in ipairs(common) do
                local f = Workspace:FindFirstChild(fname)
                if f then
                    for _, child in ipairs(f:GetChildren()) do
                        if string_find(string_lower(child.Name), "egg", 1, true) then
                            table_insert(EggCache.folderRefs, child)
                        end
                    end
                end
            end
        end
    end)
end

local function FindNearestEgg(maxDist)
    if not State.hrp then return nil, math_huge end
    RescanEggFolders()
    local myPos = State.hrp.Position
    local bestEgg, bestDist = nil, maxDist or 500

    for _, folder in ipairs(EggCache.folderRefs) do
        if folder and folder.Parent then
            for _, obj in ipairs(folder:GetChildren()) do
                local primary
                if obj:IsA("BasePart") then
                    primary = obj
                elseif obj:IsA("Model") then
                    primary = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                end
                if primary then
                    local d = (primary.Position - myPos).Magnitude
                    if d < bestDist then
                        bestDist = d
                        bestEgg = primary
                    end
                end
            end
        end
    end
    return bestEgg, bestDist
end

local function FindNearestPlayer(maxRange, targetPart)
    if not State.hrp then return nil, nil end
    local myPos = State.hrp.Position
    local bestPlayer, bestPart, bestDist = nil, nil, maxRange or 300

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            local part = player.Character:FindFirstChild(targetPart or "Head")
            local hum = player.Character:FindFirstChildOfClass("Humanoid")
            if part and hum and hum.Health > 0 then
                local d = (part.Position - myPos).Magnitude
                if d < bestDist then
                    bestDist = d
                    bestPlayer = player
                    bestPart = part
                end
            end
        end
    end
    return bestPlayer, bestPart, bestDist
end

local function RunAimbot()
    local cam = Workspace.CurrentCamera
    if not cam then return end
    local _, targetPart = FindNearestPlayer(State.aimbotRange, State.aimbotTarget)
    if targetPart then
        pcall(function()
            local goal = CFrame_lookAt(cam.CFrame.Position, targetPart.Position)
            cam.CFrame = cam.CFrame:Lerp(goal, State.aimbotSmooth)
        end)
    end
end

local function RunKillAura()
    if not State.hrp then return end
    local myPos = State.hrp.Position
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            local hrp = player.Character:FindFirstChild("HumanoidRootPart")
            local hum = player.Character:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                if (hrp.Position - myPos).Magnitude <= State.killAuraRange then
                    pcall(function()
                        local tool = State.char and State.char:FindFirstChildOfClass("Tool")
                        if tool then tool:Activate() end
                    end)
                end
            end
        end
    end
end

local function SafeLoop(name, interval, fn)
    local th = task_spawn(function()
        while State.alive and not Lifecycle.destroyed do
            task_wait(interval)
            if not State.alive or Lifecycle.destroyed then break end
            local ok, err = pcall(fn)
            if not ok then Logger.err("Loop:" .. name, "exec failed", err) end
        end
        Logger.log("Loop", name .. " terminated")
    end)
    Lifecycle.TrackThread(th)
end

    if not State.autoSteal or not State.hrp then return end
    local now = os_clock()
    if now - EggCache.lastSteal < EggCache.stealCooldown then return end
    local egg, dist = FindNearestEgg(500)
    if egg and dist and dist < 500 then
        Humanizer.RandomDelay(0.05, 0.15)
        local targetPos = Humanizer.Jitter(egg.Position + Vector3_new(0, 5, 0), 2)
        Humanizer.TweenTeleport(State.hrp, targetPos, 0.15)
        task_wait(0.2)
        pcall(function()
            local prompt = egg:FindFirstChildOfClass("ProximityPrompt")
            if prompt and fireproximityprompt then
                fireproximityprompt(prompt)
            end
        end)
        EggCache.lastSteal = os_clock()
    end
end)
